cmake_minimum_required(VERSION 3.24)
project(MSensorFileSource LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost 1.85 REQUIRED COMPONENTS container program_options filesystem date_time regex iostreams thread)
find_package(FairLogger 2.3.0 REQUIRED)
find_package(FairMQ 1.10.0 REQUIRED)
add_executable(MSensorFileSource MSensorFileSource.cxx)
target_include_directories(MSensorFileSource PRIVATE /opt/build/src/nestdaq-user-impl /opt/nestdaq/include /opt/nestdaq/include/fairlogger/bundled)
target_link_directories(MSensorFileSource PRIVATE /opt/nestdaq/lib)
target_link_libraries(MSensorFileSource PRIVATE FairMQ::FairMQ FairLogger::FairLogger Boost::program_options redis++ hiredis)
set_target_properties(MSensorFileSource PROPERTIES BUILD_RPATH "/opt/nestdaq/lib;/opt/conda/lib")

add_executable(MSensorStreamServer MSensorStreamServer.cxx)
target_include_directories(MSensorStreamServer PRIVATE /opt/build/src/nestdaq-user-impl /opt/nestdaq/include /opt/nestdaq/include/fairlogger/bundled)
target_link_directories(MSensorStreamServer PRIVATE /opt/nestdaq/lib)
target_link_libraries(MSensorStreamServer PRIVATE FairMQ::FairMQ FairLogger::FairLogger Boost::program_options redis++ hiredis)
set_target_properties(MSensorStreamServer PROPERTIES BUILD_RPATH "/opt/nestdaq/lib;/opt/conda/lib")

find_package(ROOT 6.30 REQUIRED COMPONENTS RIO Hist Gpad RHTTP)
add_executable(MSensorDqm MSensorDqm.cxx)
target_include_directories(MSensorDqm PRIVATE /opt/build/src/nestdaq-user-impl /opt/nestdaq/include /opt/nestdaq/include/fairlogger/bundled)
target_link_directories(MSensorDqm PRIVATE /opt/nestdaq/lib)
target_link_libraries(MSensorDqm PRIVATE FairMQ::FairMQ FairLogger::FairLogger Boost::program_options redis++ hiredis ROOT::RIO ROOT::Hist ROOT::Gpad ROOT::RHTTP)
set_target_properties(MSensorDqm PROPERTIES CXX_STANDARD 20 BUILD_RPATH "/opt/nestdaq/lib;/opt/conda/lib")
